Save as: .github/workflows/owasp-sentinel.yml 
name: OWASP Sentinel SAST Security Audit

on:
  push:
    branches: [ main, master ]
  pull_request:
    branches: [ main, master ]

jobs:
  sast-scan:
    name: OWASP Top 10 Security Scan
    runs-on: ubuntu-latest
    permissions:
      contents: read
      security-events: write # Required for uploading SARIF to GitHub Security

    steps:
      - name: Checkout Code
        uses: actions/checkout@v4

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: '3.11'

      - name: Install OWASP Sentinel
        run: |
          git clone --depth 1 https://github.com/your-org/owasp-sentinel.git /tmp/sentinel
          pip install -r /tmp/sentinel/requirements.txt

      - name: Run OWASP Sentinel SAST & Quality Gate
        run: |
          PYTHONPATH=/tmp/sentinel python /tmp/sentinel/cli.py scan . \
            --sarif sentinel-results.sarif \
            --fail-on-critical \
            --min-score 70

      - name: Upload SARIF to GitHub Advanced Security
        if: always()
        uses: github/codeql-action/upload-sarif@v3
        with:
          sarif_file: sentinel-results.sarif
Add to: .gitlab-ci.yml 
stages:
  - test
  - security

owasp_sentinel_sast:
  stage: security
  image: python:3.11-slim
  script:
    - apt-get update && apt-get install -y git
    - git clone --depth 1 https://github.com/your-org/owasp-sentinel.git /tmp/sentinel
    - pip install -r /tmp/sentinel/requirements.txt
    - PYTHONPATH=/tmp/sentinel python /tmp/sentinel/cli.py scan . --json report.json --sarif gl-sast-report.sarif --fail-on-critical --min-score 75
  artifacts:
    reports:
      sast: gl-sast-report.sarif
    paths:
      - report.json
    when: always
Add to: .pre-commit-config.yaml 
repos:
  - repo: local
    hooks:
      - id: owasp-sentinel
        name: OWASP Sentinel Pre-Commit Security Audit
        entry: python -m scanner.cli scan . --fail-on-critical --min-score 80
        language: system
        pass_filenames: false
        always_run: true

CI/CD Quality Gate CLI Reference

--fail-on-critical

Exits with status code 1 if any Critical severity vulnerability is detected.

--min-score <0-100>

Exits with status code 1 if overall repository security score falls below threshold.

--max-critical <N>

Allows at most N critical findings before terminating the build with failure.

--sarif <path.sarif>

Exports results in OASIS SARIF v2.1.0 format for GitHub / GitLab Code Scanning.
